# Core dependencies for Child Monitor Analyzer.
#
# HOW TO INSTALL
# --------------
# Production installs MUST use the hash-pinned lockfile, which makes pip reject
# any artifact whose SHA-256 does not match the reviewed release:
#
#     pip install --require-hashes --only-binary=:all: -r requirements.lock
#     pip install --no-deps -e .
#
# install.ps1 does this automatically when requirements.lock is present.
#
# Regenerate the lockfile after changing anything below:
#
#     uv pip compile requirements.txt --generate-hashes \
#         --index-url https://pypi.org/simple \
#         --extra-index-url https://download.pytorch.org/whl/xpu \
#         --index-strategy unsafe-best-match \
#         --output-file requirements.lock
#
# (pip-tools is not usable here: it imports pip internals and breaks on
# pip >= 26. uv resolves the same requirements without that coupling. The
# second index is needed for the Intel XPU torch build; --generate-hashes is
# what keeps a second index safe, because a substituted package fails its
# hash regardless of which index served it.)
#
# WHY THE UPPER BOUNDS
# --------------------
# Every constraint below is capped below the next major version. An uncapped
# range means a single upstream release -- benign or malicious -- lands in the
# next fresh install with no review. The caps turn that into a deliberate,
# reviewable bump. Lower bounds match the versions the test suite runs against.
#
# WHAT IS NOT HERE
# ----------------
# panns-inference and torchlibrosa were REMOVED and vendored into
# src/monitor/vendor/panns/. Both were last released in early 2023 by a single
# maintainer with no release attestations, and panns_inference imported
# matplotlib at module level -- dragging matplotlib and Pillow (27 open
# advisories) into the shipped binary for zero runtime benefit.
#
# For an editable install with dev tools instead:
#   pip install -e ".[dev]"

faster-whisper>=1.2.1,<2
librosa>=0.11.0,<0.12
torch>=2.11.0,<3
